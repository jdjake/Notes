\newpart{Research Problem}

\LARGE{\vspace{0.3cm}The \emph{Fourier dimension} of a set $X \subset \R^n$ is the supremum of $s > 0$ for which there exists a finite measure $\mu$ satisfying a Fourier decay bound $|\widehat{\mu}(\xi)| \lesssim |\xi|^{-s/2}$ holds. An $m$-dimensional submanifold $M$ of $\R^{m+k}$ is a \emph{Salem set} if it has Fourier dimension $m$.

\vspace{0.3cm}

This research project determines for which $m,k \geq 0$ there exists a smooth, $m$ dimensional submanifold of $\mathbb{R}^{m+k}$ which is a Salem set.}

\LARGE{ It turns out this problem is related to the \emph{Radon-Hurwitz numbers}. For an integer $m \geq 1$, if $2^{c + 4d}$ is the largest power of two dividing $m$, where $0 \leq c \leq 3$ and $d \geq 0$, then $\rho(m) = 2^c + 8d$. For each $m$, $\rho(m)$ is the maximum number of everywhere linearly independent vector fields on $S^{m-1}$. Define $\rho(a) = 0$ if $a \in \R - \Z$.

\vspace{0.2cm}

\begin{thm}
    There exists an $m$-dimensional smooth submanifold of $\mathbf{R}^{m+k}$ which is a Salem Set if and only if $k \leq \rho(m/2) + 1$. If $k > \rho(m/2) + 1$, then any smooth $m$ dimensional submanifold of $\mathbf{R}^{m+k}$ has Fourier dimension at most $m - 1/3$.
\end{thm}
}

\newpart{Why The Radon-Hurwitz Numbers?}

\resizebox{\linewidth}{!}{Let $M$ be the graph of $\phi \colon \R^m \to \R^k$. For a smooth density $\mu$ on $M$},

\LARGE{\[ \widehat{\mu}(\xi) = \int_{\R^m} a(x) e^{2 \pi i (\alpha \cdot x + \eta \cdot \phi(x))}\; dx, \]
where $\xi = (\alpha,\eta)$ for $\alpha \in \R^m$ and $\eta \in \R^k$, and $a \in C_c^\infty(\R^m)$. Stationary phase tells us we only expect $|\widehat{\mu}(\xi)| \lesssim |\xi|^{-m/2}$ if the Hessian of the phase is non-singular at each stationary point. If $H_i(x)$ is the Hessian of $\phi_i: \R^m \to \R$, then the Hessian of $\phi$ is a multiple of $\sum \eta_i H_i(x)$, which must be invertible for each $\eta \neq 0$ and each $x \in \text{supp}(a)$.

\vspace{0.2cm}

If $A_1,\dots,A_k$ are orthogonal $m \times m$ matrices s.t. $\sum \eta_i A_i$ is invertible for $\eta \neq 0$, and $A_1 = I$, then the vector fields
%
\[ X_1,\dots,X_{k-1}: S^{m-1} \to \R^m \]
%
defined by letting $X_i(x)$ be the orthogonal projection of $\sum x_i A_i$ onto $T_x S^{m-1}$ are everywhere linearly independent. Using this, in 1965 Adams, Lax, and Phillips showed that $\rho(m/2) + 1$ is the largest dimension of a subspace of $m \times m$ symmetric matrices such that any non-zero element is non-singular.

\newpart{Coverings by Parallel Tubes}

{\LARGE Suppose $M$ is an $m$-dimensional manifold in $\R^{m+k}$, where $k > \rho(m/2) + 1$, and $\mu$ is an arbitrary measure supported on $M$ (not necessarily a smooth density) with $|\widehat{\mu}(\xi)| \lesssim |\xi|^{-s/2}$.}

\vspace{0.2cm}

{\LARGE Then for each $x_0 \in M$ there exists $\xi_0 \in (T_{x_0} M)^\perp$ and $v_0 \in T_{x_0} M$ so that $x \mapsto \xi_0 \cdot x$ vanishes to first order at $x_0$ in all directions, and vanishes to second order in the direction $v_0$.}

A variant of the Esseen concentration bound implies $\mu(T) \lesssim R^{-s/2}$ for each $T$. Summing over $T$, i.e. over the $O(R^{m/3})$ balls of radius $R^{-1/3}$ and the $O(R^{(m-1)(1/2 - 1/3)})$ tubes in each ball, gives
%
\[\mu(\R^{m+k}) \lesssim R^{m/3 + (m-1)(1/2 - 1/3)-s/2}. \]}

\newpart{Variant of Esseen Concentration Inequality}

{\LARGE
Let $T$ be a tube considered in the previous section. Suppose
%
\[ \chi(x) = \chi_0(\pi_{\xi_0}(x)) \chi_1(\pi_{\xi_0}^\perp(x)) \]
%
is $L^\infty$ normalized, where $\chi_0 \geq 0$ is smooth, compactly supported, and adapted to an interval of length $R^{-1}$, and $\chi_1 \geq 0$ is Schwartz and adapted to the $R^{-1/3} \times R^{-1/2} \times \cdots \times R^{-1/2}$ tube pointing in the direction $v_0$, with compactly supported Fourier transform. Then
%
\[ \mu(T) \lesssim \left| \int \chi(x) e^{R i v_0 \cdot x} d\mu(x) \right| = \left| \int \widehat{\chi}(\xi) \widehat{\mu}(\xi - R v_0)\; d\xi \right| \lesssim R^{-s/2}. \]
}

\newpart{More Applications of Proof Technique}

{\LARGE
A \emph{hypersurface of constant nullity $k$} is a hypersurface such that at each point, exactly $k$ principal curvatures vanish. Examples include cones, cylinders, and tangent developables of curves in $\R^3$.

\begin{thm}
    If $M$ is a hypersurface of constant nullity $k$ in $\R^n$, then $M$ has Fourier dimension $n-1-k$.
\end{thm}

\begin{wrapfigure}{R}{0.3\linewidth}
\vspace{-0.55cm}
\begin{tikzpicture}[scale=2, line cap=round]
    % Parameters
    \def\n{30}          % number of tangent lines
    \def\R{0.5}         % radius of helix projection
    \def\H{15}        % vertical rise per step
    \def\L{1.2}         % tangent length

    % Draw tangent lines
    \foreach \i in {0,...,\n} {
        \pgfmathsetmacro{\t}{\i*\H}            % angle in degrees
        \pgfmathsetmacro{\x}{\R*cos(\t)}       % point on helix
        \pgfmathsetmacro{\y}{\R*sin(\t)}
        \pgfmathsetmacro{\dx}{-sin(\t)}        % tangent direction in xy
        \pgfmathsetmacro{\dy}{cos(\t)}

        % draw the tangent line segment
        \draw[line width = 0.3mm]
            (\x,\y) --
            ({\x + \L*\dx},{\y + \L*\dy});
    }

    % Draw the projected curve (optional)
    \draw[line width=0.5mm] plot[domain=0:360,samples=100]
        ({\R*cos(\x)},{\R*sin(\x)});
\end{tikzpicture}
\end{wrapfigure}


The vanishing of $k$ principal curvatures implies $M$ is foliated by $k$-dimensional planes, and upon each of these planes the normal vectors of $M$ are constant. If $\mu$ is supported on $M$ with $|\widehat{\mu}(\xi)| \lesssim |\xi|^{-s/2}$, and $T$ is an
%
\[ R^{-1} \times R^{-1/2} \times \cdots \times R^{-1/2} \times 1 \times \cdots \times 1 \]
%
plank with sidelength $R^{-1}$ normal to $M$, sidelength $1$ in the directions of the foliation, and sidelengths $R^{-1/2}$ in directions perpendicular to the foliation, then $x \mapsto \xi_0 \cdot x$ varies by $O(R^{-1})$ on $T$, where $\xi_0$ is normal to the plank, and the Esseen concentration variant implies $\mu(T) \lesssim R^{-s/2}$. Since $M$ is covered by $O(R^{\frac{n-1-k}{2}})$ such planks, we conclude that $\mu(\R^n) \lesssim R^{\frac{n-1-k -s}{2}}$. If $s > n - 1 - k$, we take $R \to \infty$ and conclude $\mu = 0$, which completes the proof.
}

\vspace{0.2cm}

{\LARGE A \emph{non-degenerate curve} in $\R^n$ is the image of $\gamma: I \to \R^n$ such that at each $t \in I$, $\gamma^{(1)}, \dots, \gamma^{(n)}$ are linearly independent.

\begin{thm}
    Such curves have Fourier dimension $2/n$.
\end{thm}

We cover a length $R^{-1/n}$ curve segment by a thickness $R^{-1}$ neighborhood $\Theta$ of a hyperplane normal to the binormal vector to the curve. Then if $\mu$ is supported on the curve and $|\widehat{\mu}(\xi)| \lesssim |\xi|^{-s/2}$, then $\mu(\Theta) \lesssim R^{-s/2}$, and summing over $O(R^{1/n})$ segments gives $\mu(\R^n) \lesssim R^{1/n - s/2}$. So if $s > 2/n$, we conclude $\mu = 0$.

}
